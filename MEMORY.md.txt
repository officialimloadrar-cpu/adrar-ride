// pricingEngine.ts - نفس معادلاتك بالحرف
export function prixCoursa(d: number): number {
  if (d <= 5) return 150;
  if (d <= 10) return 150 + 10 * (d - 5); // 7=170, 10=200
  if (d <= 20) return 200; // بودة 20
  if (d <= 30) return 200 + 7 * (d - 20); // فنوغيل 30=270
  if (d <= 80) return 270 + 6.6 * (d - 30); // زاوية 80=600
  return 600 + 11.428 * (d - 80); // رقان 150=1400
}

export function prixSeats(P: number, nb: number, isFamily: boolean, d: number) {
  if (nb === 1) return P;
  if (nb === 2) return P * 1.8333;
  if (nb === 3) return d <= 80? P * 2.6667 : P * 2.5; // ينطلق مباشرة
  if (nb === 4) return d <= 80? P * 2.9667 : P * 2.8;
  return P;
}

export function isNight(): boolean {
  const h = new Date().getHours();
  return h >= 21 || h < 6;
}
export function finalPrice(dayPrice: number): number {
  return isNight()? Math.max(250, dayPrice * 1.25) : dayPrice;
}
export function commission(price: number, zone: 'inside'|'outside'): number {
  return zone === 'inside'? price * 0.13 : price * 0.15; // 13% داخل وضواحي، 15% خارج
}

// cron/checkDebts.ts
export const checkDebts = onSchedule("every 1 minute", async () => {
  // مؤقت واحد فقط 48 ساعة من آخر طلب يتمدد كل مرة
  const drivers = await db.collection('drivers').where('debtTimerExpiresAt', '<', Date.now()).get();

  drivers.forEach(async (d) => {
    const debt = d.data().accumulatedDebt; // بالمبلغ المتراكم بالعمولة الفعلية
    if (debt >= 2000) {
      await d.ref.update({ status: 'BANNED', color: 'red' }); // أحمر 2000 محظور
    } else if (debt >= 1500) {
      await d.ref.update({ color: 'orange' });
      // رسالة: "تنبيه دينك 1500 بقي 500 ويتم حظرك - حول لـ CCP: 123456 Clé 12"
    }
  });
});

export async function searchPlaces(query: string, bias: Place, bbox: string): Promise<Place[]> {
  const res = await fetch(
    https://photon.komoot.io/api/?q=${encodeURIComponent(query)}&lat=${bias.lat}&lon=${bias.lon}&bbox=${bbox}&limit=8
  );
  const json = await res.json();
  return json.features.map((f: any) => ({
    name: f.properties.name,
    lat: f.geometry.coordinates[1],
    lon: f.geometry.coordinates[0]
  }));
}

export function filterNearby(places: Place[], center: Place, radiusKm = 50): Place[] {
  return places.filter(p => {
    const R = 6371;
    const dLat = (p.lat - center.lat) * Math.PI / 180;
    const dLon = (p.lon - center.lon) * Math.PI / 180;
    const a = Math.sin(dLat/2)**2 + Math.cos(center.lat*Math.PI/180)*Math.cos(p.lat*Math.PI/180)*Math.sin(dLon/2)**2;
    return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)) < radiusKm;
  });
}

import type { Place } from "./geocoding";

export async function getRouteDistance(origin: Place, destination: Place): Promise<number> {
  try {
    const url = https://router.project-osrm.org/route/v1/driving/${origin.lon},${origin.lat};${destination.lon},${destination.lat}?overview=false;
    const res = await fetch(url);
    const data = await res.json();
    if (data.routes?.[0]) return data.routes[0].distance / 1000;
  } catch {}

  const R = 6371;
  const dLat = (destination.lat - origin.lat) * Math.PI / 180;
  const dLon = (destination.lon - origin.lon) * Math.PI / 180;
  const a = Math.sin(dLat/2)**2 + Math.cos(origin.lat*Math.PI/180)*Math.cos(destination.lat*Math.PI/180)*Math.sin(dLon/2)**2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

export async function getRegions(countryCode: string): Promise<string[]> {
  // 1. Try cache first (data from last successful API call)
  const cached = localStorage.getItem(regions_${countryCode});
  if (cached) {
    try { return JSON.parse(cached); } catch {}
  }

  // 2. Fetch from database - single source of truth
  const res = await fetch(https://api.imrcoursa.com/regions?country=${countryCode});
  if (!res.ok) throw new Error('Failed to fetch regions');
  
  const data = await res.json();
  const names = data.map((r: any) => r.name);
  
  // 3. Save to cache for offline use
  localStorage.setItem(regions_${countryCode}, JSON.stringify(names));
  
  return names;
}
